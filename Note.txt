ลำดับการ Apply
# 1. ติดตั้ง ArgoCD
    kubectl apply -k GitOps/argocd/install/

# 2. รอ ArgoCD พร้อม
    kubectl wait --for=condition=available deployment/argocd-server -n argocd --timeout=120s

# 3. สร้าง Project
    kubectl apply -f GitOps/argocd/projects/sample-project.yaml

# 4. Apply Applications (ทำทุกครั้งที่เพิ่ม app ใหม่)
    kubectl apply -f GitOps/argocd/applications/dev/app1/sample-app.yaml
    kubectl apply -f GitOps/argocd/applications/prod/app1/sample-app.yaml

เพิ่ม App ใหม่
1. สร้าง GitOps/apps/base/app2/  พร้อม deployment + service + kustomization
2. สร้าง GitOps/apps/overlays/dev/app2/kustomization.yaml
3. สร้าง GitOps/apps/overlays/prod/app2/kustomization.yaml
4. สร้าง GitOps/argocd/applications/dev/app2/new-service.yaml
5. สร้าง GitOps/argocd/applications/prod/app2/new-service.yaml
6. kubectl apply -f GitOps/argocd/applications/dev/app2/new-service.yaml
7. kubectl apply -f GitOps/argocd/applications/prod/app2/new-service.yaml
